# ==========================================================
# 1. Go to development workspace
# ==========================================================

cd ~/development

# ==========================================================
# 2. Create the real Flutter project
# ==========================================================

flutter create flutter_status_saver

# ==========================================================
# 3. Enter project
# ==========================================================

cd ~/development/flutter_status_saver

# ==========================================================
# 4. Verify
# ==========================================================

pwd
flutter --version
dart --version
flutter doctor -v

# ==========================================================
# 5. Git
# ==========================================================

git init
git status

# ==========================================================
# 6. Dependencies baseline
# ==========================================================

flutter pub get

# ==========================================================
# 7. Static analysis
# ==========================================================

flutter analyze

# ==========================================================
# 8. Tests
# ==========================================================

flutter test

# ==========================================================
# 9. Check devices
# ==========================================================

flutter devices

# ==========================================================
# 10. Run Android
# ==========================================================

flutter emulators --launch Pixel_9_Pro

# Wait for emulator to boot, then:
flutter devices

# Replace with the actual Android device ID:
flutter run -d <ANDROID_DEVICE_ID>

# ==========================================================
# 11. Build debug APK
# ==========================================================

flutter build apk --debug

# ==========================================================
# 12. Git baseline
# ==========================================================

git add .
git commit -m "chore: bootstrap Flutter app"
